#!/bin/bash # ============================================================================= # 回归测试报告生成器 (Python版) # 读取JSON报告并生成HTML格式 # ============================================================================= # 使用方法: # ./regression-report.html [json_file] # # 默认读取 /tmp/regression-report-*.json 最新文件 # ============================================================================= set -euo pipefail # 查找最新报告文件 if [[ -n "${1:-}" && -f "$1" ]]; then REPORT_FILE="$1" elif [[ -d "/tmp" ]]; then REPORT_FILE=$(ls -t /tmp/regression-report-*.json 2>/dev/null | head -1 || echo "") else REPORT_FILE="" fi if [[ -z "$REPORT_FILE" || ! -f "$REPORT_FILE" ]]; then echo "错误: 未找到回归测试报告文件" echo "请先运行: ./regression-test.sh" exit 1 fi OUT_FILE="${REPORT_FILE%.json}.html" python3 << PYEOF import json import sys from datetime import datetime try: with open("$REPORT_FILE") as f: data = json.load(f) except (FileNotFoundError, json.JSONDecodeError) as e: print(f"错误: 无法读取报告文件 ($REPORT_FILE): {e}") sys.exit(1) summary = data.get('summary', {}) results = data.get('results', []) total = summary.get('total', 0) passed = summary.get('passed', 0) failed = summary.get('failed', 0) skipped = summary.get('skipped', 0) pass_rate = summary.get('pass_rate', 0) duration_ms = summary.get('duration_ms', 0) timestamp = summary.get('timestamp', '') try: dt = datetime.fromisoformat(timestamp.replace('Z', '+00:00')) date_str = dt.strftime('%Y-%m-%d %H:%M:%S') except: date_str = timestamp passed_green = passed failed_red = failed skipped_gray = skipped # Status icon mapping STATUS_ICONS = {'passed': '✅', 'failed': '❌', 'warning': '⚠️', 'skipped': '⏭️'} STATUS_LABELS = {'passed': '通过', 'failed': '失败', 'warning': '警告', 'skipped': '跳过'} STATUS_COLORS = {'passed': 'pass', 'failed': 'fail', 'warning': 'warn', 'skipped': 'skip'} STATUS_CSS = {'passed': 'rgba(16, 185, 129, 0.15)', 'failed': 'rgba(239, 68, 68, 0.15)', 'warning': 'rgba(245, 158, 11, 0.15)', 'skipped': 'rgba(107, 114, 128, 0.15)'} STATUS_TEXT_COLOR = {'passed': '#10b981', 'failed': '#ef4444', 'warning': '#f59e0b', 'skipped': '#6b7280'} rows_html = "" for r in results: name = r.get('name', 'Unknown') status = r.get('status', 'unknown') dur_ms = r.get('duration_ms', 0) message = r.get('message', '') issue = r.get('issue', '') icon = STATUS_ICONS.get(status, '❓') label = STATUS_LABELS.get(status, status) color = STATUS_COLORS.get(status, 'skip') css_bg = STATUS_CSS.get(status, 'rgba(107, 114, 128, 0.15)') css_color = STATUS_TEXT_COLOR.get(status, '#6b7280') issue_html = f'{issue}' if issue else '—' # Duration bar color if dur_ms < 200: dur_bar_color = '#10b981' elif dur_ms < 1000: dur_bar_color = '#f59e0b' else: dur_bar_color = '#ef4444' dur_bar_width = min(dur_ms / 2000 * 100, 100) rows_html += f""" {icon} {label} {name} {issue_html}
{dur_ms}ms
{message} """ html_content = f""" School Admin - 回归测试报告

🎯 School Admin - 回归测试报告

生成时间: {date_str}  |  Issue #133  |  Duration: {duration_ms}ms
{total}
测试总数
{passed_green}
通过 ✅
{failed_red}
失败 ❌
{skipped_gray}
跳过 ⏭️
{pass_rate}%
通过率
测试进度 {passed}/{total} 通过
📋 测试详情 ({len(results)} 项)
{rows_html}
状态 测试名称 关联Issue 响应时间 详情

School Admin System - 自动化回归测试 | Issue #133

防止已修复缺陷再次出现

""" with open("$OUT_FILE", "w", encoding="utf-8") as out: out.write(html_content) print(f"✅ HTML报告已生成: $OUT_FILE") print(f" 测试总数: {total} | 通过: {passed} | 失败: {failed} | 跳过: {skipped}") PYEOF
